const DataManager = {
  STORAGE_KEY: 'fouta_erp_data',
  
  saveData(key, data) {
    try {
      const allData = this.getAllData();
      allData[key] = data;
      localStorage.setItem(this.STORAGE_KEY, JSON.stringify(allData));
      return true;
    } catch (error) {
      console.error('Erreur sauvegarde:', error);
      return false;
    }
  },
  
  getData(key) {
    try {
      const allData = this.getAllData();
      return allData[key] || null;
    } catch (error) {
      console.error('Erreur lecture:', error);
      return null;
    }
  },
  
  getAllData() {
    try {
      const data = localStorage.getItem(this.STORAGE_KEY);
      return data ? JSON.parse(data) : {};
    } catch (error) {
      return {};
    }
  },
  
  clearAll() {
    localStorage.clear();
  }
};

// ==================== QR CODE GENERATOR ====================
const generateQRCode = (data) => {
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  canvas.width = 200;
  canvas.height = 200;
  
  if (ctx) {
    ctx.fillStyle = '#fff';
    ctx.fillRect(0, 0, 200, 200);
    ctx.fillStyle = '#000';
    
    // Dessiner un motif QR simplifié
    for (let i = 0; i < 10; i++) {
      for (let j = 0; j < 10; j++) {
        if ((i + j + data.length) % 2 === 0) {
          ctx.fillRect(i * 20, j * 20, 18, 18);
        }
      }
    }
    
    // Ajouter le texte des données
    ctx.fillStyle = '#666';
    ctx.font = '8px monospace';
    ctx.fillText(data.substring(0, 20), 10, 195);
  }
  
  return canvas.toDataURL();
};

// ==================== EXPORT EXCEL ====================
const exportToExcel = (data, filename) => {
  const headers = Object.keys(data[0] || {});
  const csv = [
    headers.join(','),
    ...data.map(row => headers.map(h => row[h] || '').join(','))
  ].join('\n');
  
  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = `${filename}_${new Date().toISOString().split('T')[0]}.csv`;
  link.click();
};

// ==================== IMPORT EXCEL ====================
const importFromExcel = (file) => {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const text = e.target.result;
        const lines = text.split('\n');
        const headers = lines[0].split(',');
        const data = lines.slice(1).map(line => {
          const values = line.split(',');
          const obj = {};
          headers.forEach((h, i) => obj[h.trim()] = values[i]?.trim());
          return obj;
        }).filter(row => Object.values(row).some(v => v));
        resolve(data);
      } catch (error) {
        reject(error);
      }
    };
    reader.onerror = reject;
    reader.readAsText(file);
  });
};

          };
          setNotifications(prev => [newNotif, ...prev.slice(0, 19)]);
        }
      });
    }, 30000); // Toutes les 30 secondes
    
    return () => clearInterval(interval);
  }, [machines, notifications, currentUser]);

  // ==================== HANDLERS ====================
  const handleLogin = () => {
    setLoginError('');
    const user = AuthManager.login(loginUsername, loginPassword);
    
    if (user) {
      setCurrentUser(user);
      setLoginUsername('');
      setLoginPassword('');
    } else {
      setLoginError('Identifiants incorrects');
    }
  };

  const handleLogout = () => {
    AuthManager.logout();
    setCurrentUser(null);
    setCurrentView('dashboard');
  };

  const handleDragStart = (e, of) => {
    e.dataTransfer.setData('ofId', of.id);
  };

  const handleDrop = (e, machineId, position) => {
    e.preventDefault();
    const ofId = e.dataTransfer.getData('ofId');

          id: Date.now(),
          type: 'info',
          message: `${of.id} assigné à ${machineId} position ${position}`,

          read: false

        return { ...of, machine: machineId, position, status: 'planifie' };

      return of;
    }));
  };

      id: Date.now(),
      type: 'success',
      message: `Couleurs attribuées à ${ofId}`,
      time: 'maintenant',
      read: false

  };

    setOfs(prev => prev.map(of => 
      of.id === ofId ? { ...of, status: 'en_cours' } : of

    const of = ofs.find(o => o.id === ofId);
    if (of && of.machine) {
      const machine = machines.find(m => m.id === of.machine);
      if (machine) {
        // Créer étiquette de démarrage
        const etiquette = {
          id: `ETQ-${Date.now()}`,
          ofId: ofId,
          type: 'debut_tissage',
          machine: of.machine,
          operateur: machine.operateur,
          date: new Date().toISOString(),
          qrCode: generateQRCode(`${ofId}_${of.machine}_${Date.now()}`)
        };
        setEtiquettes(prev => [...prev, etiquette]);
        
        const newNotif = {
          id: Date.now(),
          type: 'success',
          message: `${ofId} démarré sur ${of.machine}`,
          time: 'maintenant',
          read: false
        };
        setNotifications(n => [newNotif, ...n.slice(0, 19)]);
      }
    }
  };

  const handleTerminerOF = (ofId) => {
    const of = ofs.find(o => o.id === ofId);
    if (of && of.machine) {
      // Consommer l'ensouple
      setMachines(prev => prev.map(m => {
        if (m.id === of.machine) {
          const consommation = of.quantite * 2.4; // 2.4m par pièce (exemple)
          const nouveauRestant = Math.max(0, m.ensoupleRestant - consommation);
          
          // Créer alerte si critique
          if (nouveauRestant <= 500) {
            const newNotif = {
              id: Date.now(),
              type: 'alert',
              message: `⚠️ Machine ${m.id} - Ensouple critique à ${nouveauRestant.toFixed(0)}m`,
              time: 'maintenant',
              read: false
            };
            setNotifications(n => [newNotif, ...n.slice(0, 19)]);
          }
          
          return { ...m, ensoupleRestant: nouveauRestant };
        }
        return m;
      }));
      
      // Marquer OF terminé
      setOfs(prev => prev.map(o => 
        o.id === ofId ? { ...o, status: 'termine' } : o
      ));
      
      // Créer étiquette fin fabrication

        id: `ETQ-FIN-${Date.now()}`,
        ofId: ofId,
        type: 'fin_fabrication',
        machine: of.machine,

        id: Date.now(),
        type: 'success',
        message: `✅ ${ofId} terminé - Prêt pour coupe`,
        time: 'maintenant',
        read: false
      };

        id: Date.now(),
        type: 'info',
        message: `🔄 Ordissage demandé pour ${machineId} - ${poidsNecessaire.toFixed(1)}kg nécessaire`,

        read: false
      };
      setNotifications(n => [newNotif, ...n.slice(0, 19)]);

      type: 'success',
      message: `Client ${newClient.nom} ajouté`,
      time: 'maintenant',
      read: false

  };

  // Rediriger vers dashboard si accès refusé
  useEffect(() => {

        id: Date.now(),
        type: 'alert',
        message: '⚠️ Accès refusé - Module non autorisé pour votre rôle',

                onChange={(e) => setLoginUsername(e.target.value)}

              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">Mot de passe</label>
              <input

                value={loginPassword}
                onChange={(e) => setLoginPassword(e.target.value)}
                onKeyPress={(e) => e.key === 'Enter' && handleLogin()}

              <div className="p-3 bg-red-50 border border-red-200 rounded-lg flex items-center gap-2">
                <AlertCircle className="w-5 h-5 text-red-600" />

              <LogIn className="w-5 h-5" />
              Se connecter

            <div className="mt-6 p-4 bg-blue-50 rounded-lg">
              <p className="text-sm text-blue-800 font-medium mb-2">🔐 Comptes de test :</p>
              <div className="space-y-1 text-xs text-blue-700">
                <p>• <strong>admin</strong> / admin123 (Accès complet)</p>

  return (

        sidebarOpen ? 'w-64' : 'w-20'

          <nav className="space-y-1">

                className={`w-full flex items-center gap-3 px-4 py-3 rounded-lg transition-colors ${
                  currentView === item.id ? 'bg-blue-600 text-white shadow-lg' : 'text-gray-700 hover:bg-gray-100'

              >
                <item.icon className="w-5 h-5" />
                {sidebarOpen && <span className="font-medium">{item.label}</span>}

                  <span className="ml-auto px-2 py-1 text-xs rounded-full bg-red-500 text-white font-bold">
                    {item.badge}
                  </span>

              </button>
            ))}
          </nav>
        </div>
      </aside>

      {/* Header */}
      <header className="bg-white shadow-sm border-b fixed top-0 right-0 left-0 z-10" style={{ marginLeft: sidebarOpen ? '256px' : '80px' }}>
        <div className="px-4 py-3 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <button onClick={() => setSidebarOpen(!sidebarOpen)} className="p-2 hover:bg-gray-100 rounded-lg">
              <Menu className="w-5 h-5" />
            </button>
            
            <div className="flex items-center gap-2">

            <button

            >
              <Bell className="w-5 h-5 text-gray-600" />
              {unreadCount > 0 && (
                <span className="absolute top-1 right-1 w-5 h-5 bg-red-500 rounded-full text-white text-xs flex items-center justify-center font-bold">
                  {unreadCount}
                </span>
              )}
            </button>

            {showNotifications && (
              <div className="absolute right-4 top-16 w-96 bg-white rounded-lg shadow-2xl border z-50 max-h-96 overflow-y-auto">
                <div className="p-4 border-b sticky top-0 bg-white">
                  <h3 className="font-semibold">Notifications ({unreadCount})</h3>
                </div>
                {notifications.map(notif => (
                  <div

                    className={`p-4 border-b hover:bg-gray-50 cursor-pointer ${notif.read ? 'opacity-50' : ''}`}

                    <div className="flex items-start gap-3">
                      {notif.type === 'alert' ? (

                        <p className="text-xs text-gray-500 mt-1">{notif.time}</p>
                      </div>
                    </div>
                  </div>
                ))}

            )}

            {currentUser && (
              <>
                <div className="flex items-center gap-2 px-3 py-2 bg-gray-100 rounded-lg">
                  <div className={`w-8 h-8 rounded-full flex items-center justify-center text-white font-medium ${

                  }`}>
                    {currentUser.nom.charAt(0)}
                  </div>
                  <div className="hidden md:block">
                    <p className="text-sm font-medium text-gray-800">{currentUser.nom}</p>
                    <p className="text-xs text-gray-500">
                      {currentUser.role} 
                      {currentUser.role === 'Admin' && ' - Accès complet'}
                    </p>

                  <LogOut className="w-5 h-5" />

        {/* DASHBOARD */}

                  <button 
                    onClick={() => {
                      if (window.confirm('⚠️ ATTENTION : Toutes les données seront effacées définitivement. Continuer ?')) {
                        DataManager.clearAll();
                        AuthManager.logout();
                        window.location.reload();
                      }
                    }}
                    className="w-full px-4 py-3 bg-red-600 text-white rounded-lg hover:bg-red-700 flex items-center justify-center gap-2"
                  >
                    <Trash2 className="w-5 h-5" />
                    Effacer Toutes les Données (Admin uniquement)
                  </button>
                )}
              </div>
            </div>
          </div>
        )}
        {/* MESSAGE ACCÈS REFUSÉ */}
        {currentUser && !hasAccessToCurrentView() && currentView !== 'dashboard' && (
          <div className="min-h-[60vh] flex items-center justify-center">
            <div className="bg-white rounded-xl shadow-2xl p-12 max-w-md text-center">
              <div className="w-20 h-20 bg-red-100 rounded-full flex items-center justify-center mx-auto mb-6">
                <AlertCircle className="w-12 h-12 text-red-600" />
              </div>
              <h2 className="text-2xl font-bold text-gray-800 mb-4">Accès Refusé</h2>
              <p className="text-gray-600 mb-6">
                Vous n'avez pas l'autorisation d'accéder à ce module avec votre rôle <strong>{currentUser.role}</strong>.
              </p>
              <div className="space-y-3">
                <button
                  onClick={() => setCurrentView('dashboard')}
                  className="w-full px-6 py-3 bg-blue-600 text-white rounded-lg hover:bg-blue-700 font-semibold"
                >
                  Retour au Tableau de Bord
                </button>
                {currentUser.role !== 'Admin' && (
                  <p className="text-sm text-gray-500">
                    Contactez l'administrateur pour obtenir l'accès
                  </p>
                )}
              </div>
            </div>
          </div>
        )}
      </main>

      {/* MODAL ATTRIBUTION COULEURS */}
      {showPlanningModal && selectedOF && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-lg shadow-2xl max-w-md w-full p-6">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-lg font-semibold">Attribution Couleurs - {selectedOF.id}</h3>
              <button onClick={() => setShowPlanningModal(false)} className="p-2 hover:bg-gray-100 rounded">
                <X className="w-5 h-5" />
              </button>
            </div>
            
            <div className="space-y-4">
              <div>
                <label className="block text-sm font-medium mb-2">Couleur 1</label>
                <select 
                  onChange={(e) => {
                    const couleur2 = document.getElementById('couleur2').value;
                    if (e.target.value && couleur2) {
                      handleAttributionCouleurs(selectedOF.id, e.target.value, couleur2);
                      setShowPlanningModal(false);
                    }
                  }}
                  className="w-full px-3 py-2 border rounded-lg"
                >
                  <option value="">Sélectionner...</option>
                  {couleursDisponibles.map(c => (
                    <option key={c.code} value={c.code} disabled={c.stock === 0}>
                      {c.code} - {c.nom} ({c.stock}kg)
                    </option>
                  ))}
                </select>
              </div>
              
              <div>
                <label className="block text-sm font-medium mb-2">Couleur 2</label>
                <select 
                  id="couleur2"
                  className="w-full px-3 py-2 border rounded-lg"
                >
                  <option value="">Sélectionner...</option>
                  {couleursDisponibles.map(c => (
                    <option key={c.code} value={c.code} disabled={c.stock === 0}>
                      {c.code} - {c.nom} ({c.stock}kg)
                    </option>
                  ))}
                </select>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* MODAL QR CODE */}
      {showQRModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-lg shadow-2xl max-w-md w-full p-6">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-lg font-semibold">QR Code Généré</h3>
              <button onClick={() => setShowQRModal(false)} className="p-2 hover:bg-gray-100 rounded">
                <X className="w-5 h-5" />
              </button>
            </div>
            <div className="flex justify-center">
              <img src={qrData} alt="QR Code" className="w-64 h-64 border-2 border-gray-200 rounded-lg" />
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
